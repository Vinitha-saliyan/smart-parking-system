class AppConstants {
  static const String _apiOverride = String.fromEnvironment(
    'API_BASE_URL',
  );

  static const String _webApiUrl = 'http://127.0.0.1:8000/api/v1';

  static const String _physicalDeviceApiUrl = 'http://127.0.0.1:8000/api/v1';

  static String get apiBaseUrl {
    final url = _apiOverride.isNotEmpty
        ? _apiOverride
        : kIsWeb
            ? _webApiUrl
            : _physicalDeviceApiUrl;

    // Prevent accidental double slashes in API endpoints.
    return url.endsWith('/') ? url.substring(0, url.length - 1) : url;
  }

  static String get wsBaseUrl {
    return apiBaseUrl
        .replaceFirst('http://', 'ws://')
        .replaceFirst('https://', 'wss://')
        .replaceFirst('/api/v1', '/ws');
  }

  static const int reservationTimeoutMinutes = 30;

  static const Duration animationDuration = Duration(
    milliseconds: 300,
  );
}

import 'package:smart_parking_user/ui/shared/main_scaffold.dart';

final appRouterProvider = Provider<GoRouter>((ref) {
  late final GoRouter router;

  router = GoRouter(
    initialLocation: '/login',
    redirect: (context, state) {
      final authState = ref.read(authStateProvider);
      final location = state.uri.path;

      final isAuthenticated = authState is AuthAuthenticated;
      final isChecking = authState is AuthInitial || authState is AuthLoading;

      final isPublicAuthPage = location == '/login' ||
          location == '/register' ||
          location == '/forgot-password';

      // Keep the user on login/register while an authentication
      // operation is running.
      if (isChecking) {
        if (isPublicAuthPage) {
          return null;
        }

        return '/login';
      }

      // Unauthenticated users can only open authentication pages.
      if (!isAuthenticated) {
        if (isPublicAuthPage) {
          return null;
        }

        return '/login';
      }

      // Authenticated users should not return to authentication pages.
      if (isAuthenticated && (isPublicAuthPage || location == '/')) {
        return '/dashboard';
      }

      return null;
    },
    routes: [
      GoRoute(
        path: '/',
        redirect: (context, state) => '/login',
      ),
      GoRoute(
        path: '/login',
        builder: (context, state) => const LoginScreen(),
      ),
      GoRoute(
        path: '/register',
        builder: (context, state) => const RegisterScreen(),
      ),
      GoRoute(
        path: '/forgot-password',
        builder: (context, state) => const ForgotPasswordScreen(),
      ),
      GoRoute(
        path: '/notifications',
        builder: (context, state) => const NotificationsScreen(),
      ),
      GoRoute(
        path: '/payment',
        builder: (context, state) {
          final args = state.extra;
          if (args is! PaymentScreenArgs) {
            return const Scaffold(
              body: Center(
                child: Text('Payment details are missing.'),
              ),
            );
          }

          return PaymentScreen(args: args);
        },
      ),
      StatefulShellRoute.indexedStack(
        builder: (context, state, navigationShell) {
          return MainScaffold(
            navigationShell: navigationShell,
          );
        },
        branches: [
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/dashboard',
                builder: (context, state) => const DashboardScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/parking-map',
                builder: (context, state) => const ParkingMapScreen(),
                routes: [
                  GoRoute(
                    path: ':areaId',
                    builder: (context, state) {
                      return ParkingMapScreen(
                        areaId: state.pathParameters['areaId'],
                      );
                    },
                  ),
                ],
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/search',
                builder: (context, state) => const SearchScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/reservations',
                builder: (context, state) => const ReservationScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/profile',
                builder: (context, state) => const ProfileScreen(),
                routes: [
                  GoRoute(
                    path: 'settings',
                    builder: (context, state) => const SettingsScreen(),
                  ),
                  GoRoute(
                    path: 'history',
                    builder: (context, state) => const HistoryScreen(),
                  ),
                ],
              ),
            ],
          ),
        ],
      ),
    ],
  );

  // Refresh redirects without rebuilding and resetting the router.
  ref.listen(authStateProvider, (previous, next) {
    router.refresh();
  });

  ref.onDispose(router.dispose);

  return router;
});

import 'package:flutter/material.dart';
import 'package:intl/intl.dart';

extension BuildContextExtensions on BuildContext {
  ThemeData get theme => Theme.of(this);
  ColorScheme get colorScheme => theme.colorScheme;
  TextTheme get textTheme => theme.textTheme;
  Size get screenSize => MediaQuery.of(this).size;
  bool get isTablet => screenSize.width > 600;
  bool get isDarkMode => theme.brightness == Brightness.dark;
}

extension StringExtensions on String {
  String get capitalize {
    if (isEmpty) return this;
    return '${this[0].toUpperCase()}${substring(1)}';
  }

  bool get isValidEmail {
    return RegExp(r'^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$').hasMatch(this);
  }
}

extension DateTimeExtensions on DateTime {
  String get timeAgo {
    final now = DateTime.now();
    final difference = now.difference(this);

    if (difference.inDays > 365) return '${(difference.inDays / 365).floor()} years ago';
    if (difference.inDays > 30) return '${(difference.inDays / 30).floor()} months ago';
    if (difference.inDays > 0) return '${difference.inDays} days ago';
    if (difference.inHours > 0) return '${difference.inHours} hours ago';
    if (difference.inMinutes > 0) return '${difference.inMinutes} minutes ago';
    return 'Just now';
  }

  String get formatDate {
    return DateFormat('MMM dd, yyyy').format(this);
  }

  String get formatTime {
    return DateFormat('hh:mm a').format(this);
  }
}

extension DoubleExtensions on double {
  String toCurrency({String symbol = '₹'}) {
    return NumberFormat.currency(symbol: symbol, decimalDigits: 0).format(this);
  }
}

class Validators {
  static String? validateEmail(String? value) {
    if (value == null || value.isEmpty) {
      return 'Email is required';
    }
    final emailRegex = RegExp(r'^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$');
    if (!emailRegex.hasMatch(value)) {
      return 'Enter a valid email address';
    }
    return null;
  }

  static String? validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Password is required';
    }
    if (value.length < 8) {
      return 'Password must be at least 8 characters';
    }
    if (!value.contains(RegExp(r'[A-Z]'))) {
      return 'Password must contain at least one uppercase letter';
    }
    if (!value.contains(RegExp(r'[0-9]'))) {
      return 'Password must contain at least one number';
    }
    return null;
  }

  static String? validateName(String? value) {
    if (value == null || value.isEmpty) {
      return 'Name is required';
    }
    if (value.length < 2) {
      return 'Name must be at least 2 characters';
    }
    return null;
  }

  static String? validatePhone(String? value) {
    if (value == null || value.isEmpty) {
      return 'Phone number is required';
    }
    final phoneRegex = RegExp(r'^\+?[\d\s-]{10,}$');
    if (!phoneRegex.hasMatch(value)) {
      return 'Enter a valid phone number';
    }
    return null;
  }
}

import 'package:flutter/material.dart';
import 'package:smart_parking_user/core/constants/app_constants.dart';

class AnimatedCounter extends StatelessWidget {
  final num targetValue;
  final String prefix;
  final String suffix;
  final TextStyle? textStyle;
  final Duration duration;

  const AnimatedCounter({
    super.key,
    required this.targetValue,
    this.prefix = '',
    this.suffix = '',
    this.textStyle,
    this.duration = const Duration(seconds: 1),
  });

  @override
  Widget build(BuildContext context) {
    return TweenAnimationBuilder<num>(
      tween: Tween<num>(begin: 0, end: targetValue),
      duration: duration,
      curve: Curves.easeOutCubic,
      builder: (context, value, child) {
        String formattedValue = targetValue is int ? value.toInt().toString() : value.toStringAsFixed(1);
        return Text(
          '$prefix$formattedValue$suffix',
          style: textStyle,
        );
      },
    );
  }
}

import 'package:flutter/material.dart';
import 'package:smart_parking_user/core/constants/app_colors.dart';

class GradientButton extends StatefulWidget {
  final String text;
  final VoidCallback? onPressed;
  final bool isLoading;
  final IconData? icon;

  const GradientButton({
    super.key,
    required this.text,
    this.onPressed,
    this.isLoading = false,
    this.icon,
  });

  @override
  State<GradientButton> createState() => _GradientButtonState();
}

class _GradientButtonState extends State<GradientButton> with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _scaleAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 150),
    );
    _scaleAnimation = Tween<double>(begin: 1.0, end: 0.95).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTapDown: (_) => widget.onPressed != null ? _controller.forward() : null,
      onTapUp: (_) {
        if (widget.onPressed != null) {
          _controller.reverse();
          widget.onPressed!();
        }
      },
      onTapCancel: () => widget.onPressed != null ? _controller.reverse() : null,
      child: ScaleTransition(
        scale: _scaleAnimation,
        child: AnimatedContainer(
          duration: const Duration(milliseconds: 300),

import 'package:shared_preferences/shared_preferences.dart';
import 'package:smart_parking_user/data/models/user_model.dart';
import 'package:smart_parking_user/data/services/api_service.dart';

class AuthRepository {
  AuthRepository(this._apiService);

  final ApiService _apiService;

  Future<UserModel> login(String email, String password) async {
    final response = await _apiService.postForm(
      '/auth/login',
      data: {
        'username': email.trim().toLowerCase(),
        'password': password,
      },
    );
    await _saveTokens(response);
    return getCurrentUser();
  }

  Future<UserModel> register(
    String email,
    String password,
    String fullName,
    String phone,
  ) async {
    final response = await _apiService.post(
      '/auth/register',
      data: {
        'email': email.trim().toLowerCase(),
        'password': password,
        'full_name': fullName.trim(),
        'phone': phone.trim().isEmpty ? null : phone.trim(),
      },
    );
    await _saveTokens(response);
    return getCurrentUser();
  }

  Future<void> _saveTokens(dynamic response) async {
    if (response is! Map || response['access_token'] == null) {
      throw AppException('The backend returned an invalid login response');
    }
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('auth_token', response['access_token'].toString());
    if (response['refresh_token'] != null) {
      await prefs.setString(
        'auth_refresh_token',
        response['refresh_token'].toString(),
      );
    }
  }

  Future<void> logout() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('auth_token');
    await prefs.remove('auth_refresh_token');
  }

  Future<UserModel> getCurrentUser() async {
    final response = await _apiService.get('/auth/me');
    if (response is! Map) {
      throw AppException('The backend returned invalid user data');
    }
    return UserModel.fromJson(Map<String, dynamic>.from(response));
  }

  Future<void> forgotPassword(String email) async {
    await _apiService.post(
      '/auth/forgot-password',
      data: {'email': email.trim().toLowerCase()},
    );
  }

  Future<String?> getToken() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString('auth_token');
  }

  Future<bool> isLoggedIn() async {
    final token = await getToken();
    return token != null && token.isNotEmpty;
  }
}

    final response = await _apiService.get(
      '/parking/areas/$areaId',
    );

    return ParkingAreaModel.fromJson(
      _extractMap(response, 'parking area'),
    );
  }

  Future<List<ParkingSlotModel>> getSlotsByArea(
    String areaId, {
    int? floor,
  }) async {
    final query = floor == null
        ? null
        : <String, dynamic>{
            'floor': floor,
          };

    final response = await _apiService.get(
      '/parking/areas/$areaId/slots',
      queryParameters: query,
    );

    final items = _extractList(response, 'parking slots');

    return items.map((item) {
      if (item is! Map) {
        throw AppException('Invalid parking slot received');
      }

      return ParkingSlotModel.fromJson(
        Map<String, dynamic>.from(item),
      );
    }).toList();
  }

  Future<Map<String, dynamic>> getAreaSummary(
    String areaId,
  ) async {
    final response = await _apiService.get(
      '/parking/areas/$areaId/summary',
    );

    final floors = _extractList(
      response,
      'parking summary',
    ).map((item) {
      if (item is! Map) {
        throw AppException('Invalid floor summary received');
      }

      return Map<String, dynamic>.from(item);
    }).toList();

    var total = 0;
    var vacant = 0;
    var occupied = 0;
    var reserved = 0;

    for (final floor in floors) {
      total += _toInt(floor['total']);
      vacant += _toInt(floor['vacant']);
      occupied += _toInt(floor['occupied']);
      reserved += _toInt(floor['reserved']);
    }

    return {
      'floors': floors,
      'total': total,
      'vacant': vacant,
      'occupied': occupied,
      'reserved': reserved,
      'total_slots': total,
      'available_slots': vacant,
      'occupied_slots': occupied,
      'reserved_slots': reserved,
    };
  }

  List<dynamic> _extractList(
    dynamic response,
    String label,
  ) {
    if (response is List) {
      return response;
    }

    if (response is Map) {
      final data = response['data'];

      if (data is List) {
        return data;
      }

      final items = response['items'];

      if (items is List) {
        return items;
      }
    }

    throw AppException('Invalid $label response from server');
  }

  Map<String, dynamic> _extractMap(
    dynamic response,
    String label,
  ) {
    if (response is Map) {
      final data = response['data'];

      if (data is Map) {
        return Map<String, dynamic>.from(data);
      }

      return Map<String, dynamic>.from(response);
    }

    throw AppException('Invalid $label response from server');
  }

  int _toInt(dynamic value) {
    if (value is int) return value;
    if (value is num) return value.toInt();

    return int.tryParse(value?.toString() ?? '') ?? 0;
  }

import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:smart_parking_user/core/constants/app_constants.dart';

class AppException implements Exception {
  final String message;

  AppException(this.message);

  @override
  String toString() => message;
}

class ApiService {
  late final Dio _dio;

  ApiService() {
    _dio = Dio(
      BaseOptions(
        baseUrl: AppConstants.apiBaseUrl,
        connectTimeout: const Duration(seconds: 15),
        receiveTimeout: const Duration(seconds: 15),
        headers: const {'Accept': 'application/json'},
      ),
    );

    _dio.interceptors.add(
      InterceptorsWrapper(
        onRequest: (options, handler) async {
          final prefs = await SharedPreferences.getInstance();
          final token = prefs.getString('auth_token');

          if (token != null && token.isNotEmpty) {
            options.headers['Authorization'] = 'Bearer $token';
          }

          return handler.next(options);
        },
        onError: (DioException e, handler) async {
          if (e.response?.statusCode == 401 &&
              !e.requestOptions.path.endsWith('/auth/login')) {
            final prefs = await SharedPreferences.getInstance();
            await prefs.remove('auth_token');
          }

          return handler.next(e);
        },
      ),
    );
  }

  Future<dynamic> get(
    String path, {
    Map<String, dynamic>? queryParameters,
  }) async {
    try {
      final response = await _dio.get(
        path,
        queryParameters: queryParameters,
      );

      return response.data;
    } on DioException catch (e) {
      throw _handleError(e);
    }
  }

  Future<dynamic> post(
    String path, {
    dynamic data,
  }) async {
    try {
      final response = await _dio.post(
        path,
        data: data,
      );

      return response.data;
    } on DioException catch (e) {
      throw _handleError(e);
    }
  }

  // FastAPI OAuth2 login requires application/x-www-form-urlencoded
  Future<dynamic> postForm(
    String path, {
    Map<String, dynamic>? data,
  }) async {
    try {
      final response = await _dio.post(
        path,
        data: data,
        options: Options(
          contentType: Headers.formUrlEncodedContentType,
        ),
      );

      return response.data;
    } on DioException catch (e) {
      throw _handleError(e);
    }
  }

  Future<dynamic> put(
    String path, {
    dynamic data,
  }) async {
    try {
      final response = await _dio.put(
        path,
        data: data,
      );

      return response.data;
    } on DioException catch (e) {
      throw _handleError(e);
    }
  }

  Future<dynamic> delete(String path) async {
    try {
      final response = await _dio.delete(path);

      return response.data;
    } on DioException catch (e) {
      throw _handleError(e);
    }
  }

  AppException _handleError(DioException e) {
    if (e.type == DioExceptionType.connectionTimeout ||
        e.type == DioExceptionType.receiveTimeout ||
        e.type == DioExceptionType.connectionError) {
      return AppException(
        'Cannot reach ${AppConstants.apiBaseUrl}. Start the backend on port 8000.',
      );
    }

    if (e.response != null) {
      final data = e.response?.data;

      if (data is Map && data['detail'] != null) {
        return AppException(data['detail'].toString());
      }

      if (data is Map && data['message'] != null) {
        return AppException(data['message'].toString());
      }

      return AppException(
        'Request failed (${e.response?.statusCode})',
      );
    }

    return AppException('Network error');
  }
}

import 'dart:async';
import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:smart_parking_user/core/constants/app_constants.dart';
import 'package:web_socket_channel/web_socket_channel.dart';

class WebSocketService {
  final String areaId;

  WebSocketChannel? _channel;
  StreamSubscription<dynamic>? _subscription;

  final StreamController<Map<String, dynamic>> _controller =
      StreamController<Map<String, dynamic>>.broadcast();

  int _reconnectAttempts = 0;

  Timer? _reconnectTimer;
  Timer? _heartbeatTimer;

  bool _isDisposed = false;

  WebSocketService({
    required this.areaId,
  }) {
    _connect();
  }

  Stream<Map<String, dynamic>> get stream {
    return _controller.stream;
  }

  bool get isConnected {
    return _channel != null && !_isDisposed;
  }

  void _connect() {
    if (_isDisposed || _channel != null) {
      return;
    }

    try {
      final uri = Uri.parse(
        '${AppConstants.wsBaseUrl}/parking/$areaId',
      );

      debugPrint('Connecting WebSocket: $uri');

      final channel = WebSocketChannel.connect(uri);
      _channel = channel;

      _subscription = channel.stream.listen(
        (message) {
          _reconnectAttempts = 0;

          final rawMessage = message.toString();

          // FastAPI heartbeat response is plain text.
          if (rawMessage == 'pong') {
            debugPrint('WebSocket heartbeat received');
            return;
          }

          try {
            final decoded = jsonDecode(rawMessage);

            if (decoded is Map) {
              if (!_controller.isClosed) {
                _controller.add(
                  Map<String, dynamic>.from(decoded),
                );
              }
            } else {
              debugPrint(
                'WebSocket ignored non-map message: $decoded',
              );
            }
          } catch (error) {
            debugPrint(
              'WebSocket message parse error: $error',
            );
          }
        },
        onDone: _handleDisconnect,
        onError: (Object error) {
          debugPrint('WebSocket error: $error');
          _handleDisconnect();
        },
        cancelOnError: false,
      );

      _startHeartbeat();
    } catch (error) {
      debugPrint('WebSocket connection error: $error');
      _handleDisconnect();
    }
  }

  void _handleDisconnect() {
    _subscription?.cancel();
    _subscription = null;

    _channel?.sink.close();
    _channel = null;

    _heartbeatTimer?.cancel();
    _heartbeatTimer = null;

    if (!_isDisposed) {
      _scheduleReconnect();
    }
  }

  void _scheduleReconnect() {
    if (_isDisposed || (_reconnectTimer?.isActive ?? false)) {
      return;
    }

    const backoffSeconds = [1, 2, 4, 8, 30];

    final index = _reconnectAttempts < backoffSeconds.length
        ? _reconnectAttempts
        : backoffSeconds.length - 1;

    final delay = backoffSeconds[index];
    _reconnectAttempts++;

    debugPrint(
      'Reconnecting WebSocket in $delay seconds',
    );

    _reconnectTimer = Timer(
      Duration(seconds: delay),
      () {
        _reconnectTimer = null;
        _connect();
      },
    );
  }

  void _startHeartbeat() {
    _heartbeatTimer?.cancel();

    _heartbeatTimer = Timer.periodic(
      const Duration(seconds: 30),
      (timer) {
        if (!isConnected) {
          return;
        }

        try {
          // FastAPI expects plain-text "ping".
          _channel?.sink.add('ping');
        } catch (error) {
          debugPrint(
            'WebSocket heartbeat error: $error',
          );

          _handleDisconnect();
        }
      },
    );
  }

  Future<void> disconnect() async {
    if (_isDisposed) {
      return;
    }

    _isDisposed = true;

    _reconnectTimer?.cancel();
    _reconnectTimer = null;

    _heartbeatTimer?.cancel();
    _heartbeatTimer = null;

    await _subscription?.cancel();
    _subscription = null;

    await _channel?.sink.close();
    _channel = null;

    if (!_controller.isClosed) {
      await _controller.close();
    }
  }
}

sealed class AuthState {

class AuthInitial extends AuthState {
  const AuthInitial();
}

class AuthLoading extends AuthState {
  const AuthLoading();
}

class AuthAuthenticated extends AuthState {
  const AuthAuthenticated(this.user);

  final UserModel user;
}

class AuthUnauthenticated extends AuthState {
  const AuthUnauthenticated();

class AuthError extends AuthState {
  const AuthError(this.message);

class AuthNotifier extends StateNotifier<AuthState> {
  AuthNotifier(this._repository) : super(const AuthInitial()) {
    checkAuthStatus();
  }

  final AuthRepository _repository;

  Future<void> checkAuthStatus() async {
    try {
      if (!await _repository.isLoggedIn()) {
        state = const AuthUnauthenticated();
        return;
      }
      final user = await _repository.getCurrentUser();
      state = AuthAuthenticated(user);
    } catch (_) {
      await _repository.logout();
      state = const AuthUnauthenticated();
    }
  }

  Future<bool> login(String email, String password) async {
    state = const AuthLoading();
    try {
      final user = await _repository.login(email, password);
      state = AuthAuthenticated(user);
      return true;
    } catch (error) {
      state = AuthError(_cleanMessage(error));
      return false;
    }
  }

  Future<bool> register(
    String email,
    String password,
    String fullName,
    String phone,
  ) async {
    state = const AuthLoading();
    try {
      final user = await _repository.register(
        email,
        password,
        fullName,
        phone,
      );
      state = AuthAuthenticated(user);
      return true;
    } catch (error) {
      state = AuthError(_cleanMessage(error));
      return false;
    }
  }

  Future<void> logout() async {
    await _repository.logout();
    state = const AuthUnauthenticated();
  }

  String _cleanMessage(Object error) {
    return error
        .toString()
        .replaceFirst('Exception: ', '')
        .replaceFirst('Bad state: ', '');
  }
}

    final prefs = await SharedPreferences.getInstance();
    final isDark = prefs.getBool('isDarkMode');
    if (isDark != null) {
      state = isDark ? ThemeMode.dark : ThemeMode.light;
    }
  }

    state = state == ThemeMode.light ? ThemeMode.dark : ThemeMode.light;
    final prefs = await SharedPreferences.getInstance();
    await prefs.setBool('isDarkMode', state == ThemeMode.dark);
  }

  Future<void> setDark() async {
    state = ThemeMode.dark;
    final prefs = await SharedPreferences.getInstance();
    await prefs.setBool('isDarkMode', true);
  }

  Future<void> setLight() async {
    state = ThemeMode.light;
    final prefs = await SharedPreferences.getInstance();
    await prefs.setBool('isDarkMode', false);
  }
}

class AuthTextField extends StatefulWidget {
  final String label;
  final IconData prefixIcon;
  final bool isPassword;
  final TextEditingController? controller;
  final String? Function(String?)? validator;
  final TextInputType keyboardType;

  const AuthTextField({
    super.key,
    required this.label,
    required this.prefixIcon,
    this.isPassword = false,
    this.controller,
    this.validator,
    this.keyboardType = TextInputType.text,
  });

  @override
  State<AuthTextField> createState() => _AuthTextFieldState();
}

class _AuthTextFieldState extends State<AuthTextField> {
  bool _obscureText = true;
  final FocusNode _focusNode = FocusNode();
  bool _isFocused = false;

  @override
  void initState() {
    super.initState();
    _focusNode.addListener(() {
      setState(() {
        _isFocused = _focusNode.hasFocus;
      });
    });
  }

  @override
  void dispose() {
    _focusNode.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: widget.controller,
      focusNode: _focusNode,
      obscureText: widget.isPassword && _obscureText,
      validator: widget.validator,
      keyboardType: widget.keyboardType,
      style: context.textTheme.bodyLarge,
      decoration: InputDecoration(
        labelText: widget.label,
        labelStyle: TextStyle(
          color: _isFocused ? AppColors.primary : Colors.grey,
        ),
        prefixIcon: Icon(
          widget.prefixIcon,
          color: _isFocused ? AppColors.primary : Colors.grey,
        ),
        suffixIcon: widget.isPassword
            ? IconButton(
                icon: Icon(
                  _obscureText ? Icons.visibility_off : Icons.visibility,
                  color: Colors.grey,
                ),
                onPressed: () {
                  setState(() {
                    _obscureText = !_obscureText;
                  });
                },
              )
            : null,
      ),
    );
  }
}

class OccupancyChart extends StatefulWidget {
  final int vacant;
  final int occupied;
  final int reserved;

  const OccupancyChart({
    super.key,
    required this.vacant,
    required this.occupied,
    required this.reserved,
  });

  @override
  State<OccupancyChart> createState() => _OccupancyChartState();
}

class _OccupancyChartState extends State<OccupancyChart> {
  int touchedIndex = -1;

  @override
  Widget build(BuildContext context) {
    int total = widget.vacant + widget.occupied + widget.reserved;
    if (total == 0) total = 1; // Prevent division by zero

    return AspectRatio(
      aspectRatio: 1,
      child: PieChart(
        PieChartData(
          pieTouchData: PieTouchData(
            touchCallback: (FlTouchEvent event, pieTouchResponse) {
              setState(() {
                if (!event.isInterestedForInteractions ||
                    pieTouchResponse == null ||
                    pieTouchResponse.touchedSection == null) {
                  touchedIndex = -1;
                  return;
                }
                touchedIndex = pieTouchResponse.touchedSection!.touchedSectionIndex;
              });
            },
          ),
          borderData: FlBorderData(show: false),
          sectionsSpace: 2,
          centerSpaceRadius: 30,
          sections: [
            PieChartSectionData(
              color: AppColors.slotVacant,
              value: widget.vacant.toDouble(),
              title: '${((widget.vacant / total) * 100).toStringAsFixed(0)}%',
              radius: touchedIndex == 0 ? 40.0 : 30.0,
              titleStyle: const TextStyle(fontSize: 12, fontWeight: FontWeight.bold, color: Colors.white),
            ),
            PieChartSectionData(
              color: AppColors.slotOccupied,
              value: widget.occupied.toDouble(),
              title: '${((widget.occupied / total) * 100).toStringAsFixed(0)}%',
              radius: touchedIndex == 1 ? 40.0 : 30.0,
              titleStyle: const TextStyle(fontSize: 12, fontWeight: FontWeight.bold, color: Colors.white),
            ),
            PieChartSectionData(
              color: AppColors.slotReserved,
              value: widget.reserved.toDouble(),

class _ParkingMapScreenState extends ConsumerState<ParkingMapScreen> {
  int _selectedFloor = 1;
  bool _isReserving = false;

  String _normalizeStatus(String status) {
    return status.split('.').last.toUpperCase();
  }

  ParkingAreaModel? _findSelectedArea(
    List<ParkingAreaModel> areas,
  ) {
    if (areas.isEmpty) {
      return null;
    }

    // When opened from the bottom navigation, use the first area.
    if (widget.areaId == null) {
      return areas.first;
    }

    for (final area in areas) {
      if (area.id == widget.areaId) {
        return area;
      }
    }

    return null;
  }

  void _showMessage(
    String message, {
    Color backgroundColor = AppColors.error,
  }) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(message),
          backgroundColor: backgroundColor,
        ),
      );
  }

  void _showSlotDetails(
    ParkingSlotModel slot,
    ParkingAreaModel area,
  ) {
    showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (bottomSheetContext) {
        return StatefulBuilder(
          builder: (context, setModalState) {
            return SlotDetailSheet(
              slot: slot,
              areaName: area.name,
              hourlyRate: area.hourlyRate,
              currency: area.currency,
              isReserving: _isReserving,
              onReserve: () async {
                if (_isReserving) {
                  return;
                }

                if (_normalizeStatus(slot.status) != 'VACANT') {
                  _showMessage(
                    'This parking slot is no longer available.',
                  );
                  return;
                }

                if (mounted) {
                  setState(() {
                    _isReserving = true;
                  });
                }

                if (bottomSheetContext.mounted) {
                  setModalState(() {});
                }

                try {
                  final repository = ref.read(
                    reservationRepositoryProvider,
                  );

                  final startTime = DateTime.now();
                  final endTime = startTime.add(
                    const Duration(hours: 1),
                  );

                  final reservation = await repository.createReservation(
                    slot.id,
                    area.id,
                    startTime,
                    endTime,
                  );

                  if (!mounted) {
                    return;
                  }

                  // Refresh all data affected by the reservation.
                  ref.invalidate(activeReservationProvider);
                  ref.invalidate(reservationsListProvider);
                  ref.invalidate(
                    parkingSlotsProvider(area.id),
                  );
                  ref.invalidate(parkingAreasProvider);

                  if (bottomSheetContext.mounted) {
                    Navigator.of(bottomSheetContext).pop();
                  }

                  if (!mounted) {
                    return;
                  }

                  _showMessage(
                    'Slot held. Pay the ₹10 advance within 30 minutes.',
                    backgroundColor: AppColors.success,
                  );

                  await context.push(
                    '/payment',
                    extra: PaymentScreenArgs(
                      reservationId: reservation.id,
                      amountPaise:
                          ((reservation.totalAmount ?? 10) * 100)
                              .round(),
                      currency: 'INR',
                    ),
                  );
                } catch (error) {
                  if (!mounted) {
                    return;
                  }

                  _showMessage(error.toString());
                } finally {
                  if (mounted) {
                    setState(() {
                      _isReserving = false;
                    });
                  }

                  if (bottomSheetContext.mounted) {
                    setModalState(() {});
                  }
                }
              },
            );
          },
        );
      },
    ).whenComplete(() {
      if (mounted && _isReserving) {
        setState(() {
          _isReserving = false;
        });
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    final areasAsync = ref.watch(parkingAreasProvider);

    return areasAsync.when(
      data: (areas) {
        final selectedArea = _findSelectedArea(areas);

        if (selectedArea == null) {
          return Scaffold(
            appBar: _buildAppBar(),
            body: Center(
              child: Text(
                areas.isEmpty

                    : 'Parking area not found',

          );

      loading: () {
        return Scaffold(
          appBar: _buildAppBar(),
          body: const Center(
            child: CircularProgressIndicator(),
          ),
        );
      },
      error: (error, stackTrace) {
        return Scaffold(
          appBar: _buildAppBar(),

              padding: const EdgeInsets.all(24),
              child: Column(

                    label: const Text('Try Again'),

                ],
              ),
            ),
          ),
        );
      },
    );
  }

  PreferredSizeWidget _buildAppBar() {
    return AppBar(
      title: const Text('Parking Map'),
      leading: widget.areaId != null
          ? IconButton(
              icon: const Icon(Icons.arrow_back),
              onPressed: () {
                if (context.canPop()) {
                  context.pop();
                } else {
                  context.go('/dashboard');
                }
              },
            )
          : null,
    );
  }

  Widget _buildMap(ParkingAreaModel area) {
    final slotsAsync = ref.watch(
      parkingSlotsProvider(area.id),
    );

    final webSocketAsync = ref.watch(
      webSocketProvider(area.id),
    );

    final totalFloors = area.totalFloors > 0 ? area.totalFloors : 1;

    final currentFloor = _selectedFloor.clamp(1, totalFloors).toInt();

    return Scaffold(
      appBar: _buildAppBar(),
      body: Column(
        children: [
          FloorSelector(
            totalFloors: totalFloors,
            selectedFloor: currentFloor,
            onFloorSelected: (floor) {
              setState(() {
                _selectedFloor = floor;
              });
            },
          ),
          Expanded(
            child: slotsAsync.when(
              data: (slots) {
                final floorSlots = slots
                    .where(
                      (slot) => slot.floor == currentFloor,
                    )
                    .toList();

                if (floorSlots.isEmpty) {
                  return Center(
                    child: Text(
                      'No parking slots available on '
                      'floor $currentFloor',
                    ),
                  );
                }

                return InteractiveViewer(
                  minScale: 0.5,
                  maxScale: 3,
                  boundaryMargin: const EdgeInsets.all(100),
                  child: GridView.builder(
                    padding: const EdgeInsets.all(24),
                    gridDelegate:
                        const SliverGridDelegateWithFixedCrossAxisCount(
                      crossAxisCount: 4,
                      childAspectRatio: 0.7,
                      crossAxisSpacing: 16,
                      mainAxisSpacing: 16,
                    ),
                    itemCount: floorSlots.length,
                    itemBuilder: (context, index) {
                      final slot = floorSlots[index];

                      var currentStatus = _normalizeStatus(slot.status);

                      webSocketAsync.whenData((data) {
                        final eventSlotId = data['slot_id']?.toString();

                        final isSameSlot = eventSlotId == slot.id ||
                            eventSlotId == slot.slotId;

                        if (isSameSlot && data['status'] != null) {
                          currentStatus = _normalizeStatus(
                            data['status'].toString(),
                          );
                        }
                      });

                      return ParkingSlotWidget(
                        slotId: slot.slotId,
                        status: DomainSlotStatus.fromString(
                          currentStatus,
                        ),
                        type: DomainSlotType.fromString(
                          slot.slotType,
                        ),
                        onTap: () {
                          if (currentStatus != 'VACANT') {
                            _showMessage(
                              'Only vacant parking slots '
                              'can be booked.',
                            );
                            return;
                          }

                          _showSlotDetails(slot, area);
                        },
                      );
                    },
                  ),
                );
              },
              loading: () {
                return ShimmerLoading.grid(
                  crossAxisCount: 4,
                  childAspectRatio: 0.7,
                );
              },
              error: (error, stackTrace) {
                return Center(
                  child: Padding(
                    padding: const EdgeInsets.all(24),
                    child: Column(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        Text(
                          error.toString(),
                          textAlign: TextAlign.center,
                          style: const TextStyle(
                            color: AppColors.error,
                          ),
                        ),
                        const SizedBox(height: 16),
                        ElevatedButton.icon(
                          onPressed: () {
                            ref.invalidate(
                              parkingSlotsProvider(area.id),
                            );
                          },
                          icon: const Icon(Icons.refresh),
                          label: const Text('Try Again'),
                        ),
                      ],
                    ),
                  ),
                );
              },
            ),
          ),
          const LegendBar(),
        ],
      ),
    );
  }
}

class _PaymentScreenState extends ConsumerState<PaymentScreen> {

  PaymentOrderModel? _activeOrder;
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    _razorpay = Razorpay();
    _razorpay.on(Razorpay.EVENT_PAYMENT_SUCCESS, _handlePaymentSuccess);
    _razorpay.on(Razorpay.EVENT_PAYMENT_ERROR, _handlePaymentError);

  Future<void> _startCheckout() async {
    if (widget.args.amountPaise < 100) {
      _showMessage('Payment amount must be at least ₹1.', isError: true);
      return;
    }

    setState(() => _isLoading = true);
    try {
      final order = await ref.read(paymentRepositoryProvider).createOrder(
            reservationId: widget.args.reservationId,
            amount: widget.args.amountPaise,
            currency: widget.args.currency,
            purpose: widget.args.purpose,
          );
      if (!mounted) return;

      _activeOrder = order;
      final user = ref.read(currentUserProvider);
      final options = <String, dynamic>{
        'key': order.keyId,
        'order_id': order.orderId,
        'amount': order.amount,
        'currency': order.currency,
        'name': 'Smart Parking',
        'description': widget.args.isSettlement
            ? 'Final parking balance'
            : '₹10 reservation advance',
        'prefill': {
          'email': user?.email ?? '',
          'contact': user?.phone ?? '',
        },
        'retry': {'enabled': true, 'max_count': 3},
        'theme': {'color': '#2563EB'},
      };

      setState(() => _isLoading = false);
      _razorpay.open(options);
    } catch (error) {
      if (!mounted) return;
      setState(() => _isLoading = false);
      _showMessage(error.toString(), isError: true);
    }
  }

  Future<void> _handlePaymentSuccess(PaymentSuccessResponse response) async {
    final paymentId = response.paymentId;
    final orderId = response.orderId ?? _activeOrder?.orderId;
    final signature = response.signature;
    if (paymentId == null || orderId == null || signature == null) {
      _showMessage(
        'Razorpay returned incomplete payment details.',
        isError: true,
      );
      return;
    }

    if (mounted) setState(() => _isLoading = true);
    try {
      await ref.read(paymentRepositoryProvider).verifyPayment(
            paymentId: paymentId,
            orderId: orderId,
            signature: signature,
          );
      ref.invalidate(activeReservationProvider);
      ref.invalidate(reservationsListProvider);
      if (!mounted) return;
      setState(() => _isLoading = false);
      _showMessage(
        widget.args.isSettlement
            ? 'Final payment verified. Checkout is complete.'
            : '₹10 advance verified. Your parking QR is ready.',
      );
      context.go('/reservations');
    } catch (error) {
      if (!mounted) return;
      setState(() => _isLoading = false);
      _showMessage(
        'Payment was received but verification failed: $error',
        isError: true,
      );
    }
  }

  void _handlePaymentError(PaymentFailureResponse response) {
    if (mounted) setState(() => _isLoading = false);
    final cancelled = response.code == Razorpay.PAYMENT_CANCELLED;
    final cancellationMessage = widget.args.isSettlement
        ? 'Payment cancelled. Checkout remains pending and the slot stays occupied.'
        : 'Payment cancelled. The slot remains held until the 30-minute timer expires.';
    _showMessage(
      cancelled
          ? cancellationMessage
          : 'Payment failed: ${response.message ?? 'Please try again.'}',
      isError: !cancelled,
    );
  }

  void _handleExternalWallet(ExternalWalletResponse response) {
    _showMessage(
      'External wallet selected: ${response.walletName ?? 'wallet'}',
    );
  }

  void _showMessage(String message, {bool isError = false}) {

      SnackBar(
        content: Text(message),
        backgroundColor: isError ? AppColors.error : AppColors.success,
      ),
    );
  }

class ReservationScreen extends ConsumerStatefulWidget {

  @override
  ConsumerState<ReservationScreen> createState() => _ReservationScreenState();
}

class _ReservationScreenState extends ConsumerState<ReservationScreen> {
  bool _isActionLoading = false;

  Future<void> _onRefresh() async {
    ref.invalidate(activeReservationProvider);
    ref.invalidate(reservationsListProvider);
  }

  Future<void> _cancelReservation(String id) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Cancel Reservation'),
        content: const Text(
          'Are you sure? Any ₹10 advance already paid is non-refundable.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('No'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text(
              'Yes, Cancel',
              style: TextStyle(color: AppColors.error),
            ),
          ),
        ],
      ),
    );

    if (confirmed != true) return;

    try {
      await ref.read(reservationRepositoryProvider).cancelReservation(id);
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Reservation cancelled')),
      );
      await _onRefresh();
    } catch (error) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(error.toString()),
          backgroundColor: AppColors.error,
        ),
      );
    }
  }

  Future<void> _startParking(ReservationModel reservation) async {
    if (_isActionLoading) return;
    setState(() => _isActionLoading = true);

    try {
      await ref.read(reservationRepositoryProvider).checkIn(reservation.id);
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Parking timer started from 00:00:01.'),
          backgroundColor: AppColors.success,
        ),
      );
      await _onRefresh();
    } catch (error) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(error.toString()),
          backgroundColor: AppColors.error,
        ),
      );
    } finally {
      if (mounted) setState(() => _isActionLoading = false);
    }
  }

  Future<void> _beginCheckout(ReservationModel reservation) async {
    if (_isActionLoading) return;
    setState(() => _isActionLoading = true);

    try {
      final checkout =
          await ref.read(reservationRepositoryProvider).checkOut(reservation.id);
      final totalAmount = checkout.totalAmount ?? 20;
      final amountDuePaise = ((totalAmount - 10) * 100).round();

      ref.invalidate(activeReservationProvider);
      ref.invalidate(reservationsListProvider);
      if (!mounted) return;
      setState(() => _isActionLoading = false);

      await context.push(
        '/payment',
        extra: PaymentScreenArgs(
          reservationId: checkout.id,
          amountPaise: amountDuePaise,
          purpose: 'SETTLEMENT',
        ),
      );
      if (!mounted) return;
      await _onRefresh();
    } catch (error) {
      if (!mounted) return;
      setState(() => _isActionLoading = false);
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(error.toString()),
          backgroundColor: AppColors.error,
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    final activeReservation = ref.watch(activeReservationProvider);
    final reservationHistory = ref.watch(reservationsListProvider);

    return Scaffold(
      appBar: AppBar(
        title: const Text('My Bookings'),
        centerTitle: true,
      ),
      body: RefreshIndicator(
        onRefresh: _onRefresh,
        child: ListView(
          padding: const EdgeInsets.all(20),
          children: [
            activeReservation.when(
              data: (reservation) {
                if (reservation == null) {

import 'package:smart_parking_user/core/constants/app_colors.dart';
import 'package:smart_parking_user/core/constants/app_constants.dart';

class CountdownTimer extends StatefulWidget {
  final DateTime targetTime;
  final VoidCallback? onExpire;

  const CountdownTimer({
    super.key,
    required this.targetTime,
    this.onExpire,
  });

  @override
  State<CountdownTimer> createState() => _CountdownTimerState();
}

class _CountdownTimerState extends State<CountdownTimer> {
  Timer? _timer;
  Duration _remaining = Duration.zero;
  final Duration _totalDuration = const Duration(
    minutes: AppConstants.reservationTimeoutMinutes,
  );

  @override
  void initState() {
    super.initState();
    _updateTime();
    if (_remaining > Duration.zero) {
      _timer = Timer.periodic(const Duration(seconds: 1), (_) => _updateTime());
    }
  }

  void _updateTime() {
    final now = DateTime.now();
    setState(() {
      _remaining = widget.targetTime.difference(now);
      if (_remaining.isNegative) {
        _remaining = Duration.zero;
        _timer?.cancel();
        widget.onExpire?.call();
      }
    });
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final percentage = (_remaining.inSeconds / _totalDuration.inSeconds).clamp(0.0, 1.0);
    final color = _remaining.inMinutes > 5
        ? AppColors.success
        : (_remaining.inMinutes > 2 ? AppColors.warning : AppColors.error);

    final minutes = _remaining.inMinutes.toString().padLeft(2, '0');
    final seconds = (_remaining.inSeconds % 60).toString().padLeft(2, '0');

    return CircularPercentIndicator(
      radius: 60.0,
      lineWidth: 10.0,
      percent: percentage,
      center: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Text(
            '$minutes:$seconds',
            style: TextStyle(
              fontSize: 24,
              fontWeight: FontWeight.bold,
              color: color,
            ),
          ),
          const Text(
            'Remaining',
            style: TextStyle(
              fontSize: 12,
              color: Colors.grey,
            ),
          ),
        ],
      ),
      progressColor: color,
      backgroundColor: color.withOpacity(0.2),
      circularStrokeCap: CircularStrokeCap.round,
      animation: true,
      animateFromLastPercent: true,
    );
  }
}

class _ParkingElapsedTimerState extends State<ParkingElapsedTimer> {
  Timer? _timer;
  int _elapsedSeconds = 1;

  @override
  void initState() {
    super.initState();
    _updateElapsedTime();
    _timer = Timer.periodic(
      const Duration(seconds: 1),
      (_) => _updateElapsedTime(),
    );
  }

  @override
  void didUpdateWidget(covariant ParkingElapsedTimer oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.checkInTime != widget.checkInTime) {
      _updateElapsedTime();
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }

  void _updateElapsedTime() {
    final seconds = DateTime.now().difference(widget.checkInTime).inSeconds;
    if (!mounted) return;
    setState(() => _elapsedSeconds = seconds < 1 ? 1 : seconds);
  }

  int get _totalFee {
    if (_elapsedSeconds <= 3600) return 20;
    final extraStartedHours = (_elapsedSeconds - 3600 + 3599) ~/ 3600;
    return 20 + (extraStartedHours * 5);
  }

  String get _formattedElapsed {
    final hours = _elapsedSeconds ~/ 3600;
    final minutes = (_elapsedSeconds % 3600) ~/ 60;
    final seconds = _elapsedSeconds % 60;
    return '${hours.toString().padLeft(2, '0')}:'
        '${minutes.toString().padLeft(2, '0')}:'
        '${seconds.toString().padLeft(2, '0')}';
  }

  @override
  Widget build(BuildContext context) {
    final checkoutBalance = _totalFee - 10;

    return Card(

    switch (_selectedFilter) {

      case 'EV Charging':
        return 'EV';
      case 'Accessible':
        return 'DISABLED';
      default:
        return null;
    }

  String _normalizeValue(String value) {
    return value.split('.').last.toUpperCase();
  }

  bool _hasAvailableSlotOfType(
    AsyncValue<List<ParkingSlotModel>>? slotsAsync,
    String requiredType,
  ) {
    if (slotsAsync == null) {
      return false;
    }

    return slotsAsync.maybeWhen(
      data: (slots) {
        return slots.any((slot) {
          final slotType = _normalizeValue(slot.slotType);
          final slotStatus = _normalizeValue(slot.status);

          return slotType == requiredType && slotStatus == 'VACANT';
        });
      },
      orElse: () => false,
    );
  }

  bool _matchesSearch(
    ParkingAreaModel area,
    String query,
  ) {
    if (query.isEmpty) {
      return true;
    }

    return area.name.toLowerCase().contains(query) ||
        area.address.toLowerCase().contains(query);
  }

  Future<void> _refreshAreas(
    List<ParkingAreaModel> areas,
  ) async {
    ref.invalidate(parkingAreasProvider);

      ref.invalidate(parkingSlotsProvider(area.id));
    }

                _selectedFilter = value;
              });
            },
          ).animate().fadeIn(delay: 100.ms),
          const SizedBox(height: 20),
          Expanded(
            child: areasAsync.when(
              data: (areas) {
                final requiredType = _requiredSlotType();

                final slotStates =
                    <String, AsyncValue<List<ParkingSlotModel>>>{};

                // Slot data is loaded only for filters that require it.
                if (requiredType != null) {
                  for (final area in areas) {
                    slotStates[area.id] = ref.watch(
                      parkingSlotsProvider(area.id),
                    );
                  }
                }

                final isLoadingSlots = requiredType != null &&
                    slotStates.values.any(
                      (state) => state.isLoading,
                    );

                if (isLoadingSlots) {
                  return ListView.builder(
                    padding: const EdgeInsets.symmetric(
                      horizontal: 20,
                    ),
                    itemCount: 3,
                    itemBuilder: (context, index) {
                      return Padding(
                        padding: const EdgeInsets.only(
                          bottom: 16,
                        ),
                        child: ShimmerLoading.card(
                          height: 230,
                        ),
                      );
                    },
                  );
                }

                final query = _searchController.text.trim().toLowerCase();

                final filteredAreas = areas.where((area) {
                  if (!_matchesSearch(area, query)) {
                    return false;
                  }

                  if (_selectedFilter == 'Available') {
                    return area.availableSlots > 0;
                  }

                  if (requiredType != null) {
                    return _hasAvailableSlotOfType(
                      slotStates[area.id],
                      requiredType,
                    );
                  }

                  return true;
                }).toList();

                if (filteredAreas.isEmpty) {
                  return RefreshIndicator(
                    onRefresh: () => _refreshAreas(areas),
                    child: ListView(
                      physics: const AlwaysScrollableScrollPhysics(),
                      children: const [
                        SizedBox(height: 160),
                        Center(
                          child: Text(
                            'No parking areas found.',
                          ),
                        ),
                      ],
                    ),
                  );
                }

                return RefreshIndicator(
                  onRefresh: () => _refreshAreas(areas),
                  child: ListView.builder(
                    physics: const AlwaysScrollableScrollPhysics(),
                    padding: const EdgeInsets.symmetric(
                      horizontal: 20,
                    ),
                    itemCount: filteredAreas.length,
                    itemBuilder: (context, index) {
                      final area = filteredAreas[index];

                      return Padding(
                        padding: const EdgeInsets.only(
                          bottom: 16,
                        ),
                        child: SizedBox(
                          width: double.infinity,
                          child: NearbyParkingCard(
                            area: area,
                            onTap: () {
                              // Switches correctly from the Search
                              // branch to the Parking Map branch.
                              context.go(
                                '/parking-map/${area.id}',
                              );
                            },
                          ),
                        ),
                      )
                          .animate()
                          .fadeIn(
                            delay: Duration(
                              milliseconds: 50 * index,
                            ),
                          )
                          .slideX(begin: 0.1, end: 0);
                    },
                  ),
                );
              },
              loading: () {
                return ListView.builder(
                  padding: const EdgeInsets.symmetric(
                    horizontal: 20,
                  ),
                  itemCount: 4,
                  itemBuilder: (context, index) {
                    return Padding(
                      padding: const EdgeInsets.only(
                        bottom: 16,
                      ),
                      child: ShimmerLoading.card(
                        height: 230,
                      ),
                    );
                  },
                );
              },
              error: (error, stackTrace) {
                return Center(
                  child: Padding(
                    padding: const EdgeInsets.all(24),
                    child: Column(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        Text(
                          error.toString(),
                          textAlign: TextAlign.center,
                          style: const TextStyle(
                            color: AppColors.error,
                          ),
                        ),
                        const SizedBox(height: 16),
                        ElevatedButton.icon(
                          onPressed: () {
                            ref.invalidate(
                              parkingAreasProvider,
                            );
                          },
                          icon: const Icon(Icons.refresh),
                          label: const Text('Try Again'),
                        ),
                      ],
                    ),
                  ),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}
