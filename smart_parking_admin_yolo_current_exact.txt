import 'package:smart_parking_admin/ui/analytics/analytics_screen.dart';

import 'package:smart_parking_admin/ui/dashboard/admin_dashboard_screen.dart';

import 'package:smart_parking_admin/ui/shared/admin_scaffold.dart';
import 'package:smart_parking_admin/ui/user_management/user_management_screen.dart';

final appRouterProvider = Provider<GoRouter>((ref) {
  final authState = ref.watch(authStateProvider);

  return GoRouter(
    initialLocation: '/dashboard',
    redirect: (context, state) {
      final onLoginPage = state.matchedLocation == '/login';
      if (authState is AuthInitial || authState is AuthLoading) {
        return onLoginPage ? null : '/login';
      }
      if (authState is AuthAuthenticated) {
        return onLoginPage ? '/dashboard' : null;
      }
      return onLoginPage ? null : '/login';
    },
    routes: [
      GoRoute(
        path: '/login',
        builder: (context, state) => const AdminLoginScreen(),
      ),
      StatefulShellRoute.indexedStack(
        builder: (context, state, navigationShell) {
          return AdminScaffold(navigationShell: navigationShell);
        },
        branches: [
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/dashboard',
                builder: (context, state) => const AdminDashboardScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/parking',
                builder: (context, state) => const ParkingManagementScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/users',
                builder: (context, state) => const UserManagementScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/reservations',
                builder: (context, state) =>
                    const ReservationManagementScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/analytics',
                builder: (context, state) => const AnalyticsScreen(),
              ),
            ],
          ),
          StatefulShellBranch(
            routes: [
              GoRoute(
                path: '/settings',
                builder: (context, state) => const AdminSettingsScreen(),
              ),
            ],
          ),
        ],
      ),
    ],
  );
});

import 'package:intl/intl.dart';

extension NumberFormatting on num {
  String get compactFormat {
    if (this >= 1000000) {
      return '${(this / 1000000).toStringAsFixed(1)}M';
    } else if (this >= 1000) {
      return '${(this / 1000).toStringAsFixed(1)}K';
    }
    return toString();
  }
  
  String get currencyFormat {
    final format = NumberFormat.currency(symbol: '₹', decimalDigits: 0);
    return format.format(this);
  }
}

import 'package:flutter/material.dart';

class DataTableWidget<T> extends StatelessWidget {
  final List<DataColumn> columns;
  final List<T> items;
  final DataRow Function(T item) rowBuilder;
  final bool isLoading;
  final String emptyMessage;

  const DataTableWidget({
    super.key,
    required this.columns,
    required this.items,
    required this.rowBuilder,
    this.isLoading = false,
    this.emptyMessage = 'No data available',
  });

  @override
  Widget build(BuildContext context) {
    if (isLoading) {
      return const Center(
        child: Padding(
          padding: EdgeInsets.all(32.0),
          child: CircularProgressIndicator(),
        ),
      );
    }

    if (items.isEmpty) {
      return Center(
        child: Padding(
          padding: const EdgeInsets.all(48.0),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Icon(Icons.inbox, size: 48, color: Theme.of(context).dividerColor),
              const SizedBox(height: 16),
              Text(emptyMessage, style: Theme.of(context).textTheme.titleMedium),
            ],
          ),
        ),
      );
    }

    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: DataTable(
        headingTextStyle: Theme.of(context).textTheme.labelMedium?.copyWith(
          fontWeight: FontWeight.bold,
        ),
        dataRowMaxHeight: 60,
        columns: columns,
        rows: items.map((e) => rowBuilder(e)).toList(),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'dart:async';

class SearchField extends StatefulWidget {
  final String hint;
  final ValueChanged<String> onChanged;
  final Duration debounceTime;

  const SearchField({
    super.key,
    this.hint = 'Search...',
    required this.onChanged,
    this.debounceTime = const Duration(milliseconds: 300),
  });

  @override
  State<SearchField> createState() => _SearchFieldState();
}

class _SearchFieldState extends State<SearchField> {
  Timer? _debounce;
  final _controller = TextEditingController();

  @override
  void dispose() {
    _debounce?.cancel();
    _controller.dispose();
    super.dispose();
  }

  void _onChanged(String value) {
    if (_debounce?.isActive ?? false) _debounce!.cancel();
    _debounce = Timer(widget.debounceTime, () {
      widget.onChanged(value);
    });
    setState(() {});
  }

  @override
  Widget build(BuildContext context) {
    return TextField(
      controller: _controller,
      onChanged: _onChanged,
      decoration: InputDecoration(
        hintText: widget.hint,
        prefixIcon: const Icon(Icons.search),
        suffixIcon: _controller.text.isNotEmpty
            ? IconButton(
                icon: const Icon(Icons.clear, size: 18),
                onPressed: () {
                  _controller.clear();
                  _onChanged('');
                },
              )
            : null,
        filled: true,
        fillColor: Theme.of(context).cardColor,
        border: OutlineInputBorder(
          borderRadius: BorderRadius.circular(8),
          borderSide: BorderSide.none,
        ),
        contentPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 0),
      ),
    );
  }
}

import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:smart_parking_admin/data/models/user_model.dart';
import 'package:smart_parking_admin/data/services/api_service.dart';

class AuthRepository {
  AuthRepository(this._apiService);

  final ApiService _apiService;

  Future<UserModel> login(String email, String password) async {
    try {
      final response = await _apiService.client.post<Map<String, dynamic>>(
        '/auth/login',
        data: {
          'username': email.trim().toLowerCase(),
          'password': password,
        },
        options: Options(contentType: Headers.formUrlEncodedContentType),
      );
      final data = response.data;
      if (data == null || data['access_token'] == null) {
        throw StateError('The backend returned an invalid login response');
      }

      final prefs = await SharedPreferences.getInstance();
      await prefs.setString('admin_token', data['access_token'] as String);
      final refreshToken = data['refresh_token'] as String?;
      if (refreshToken != null) {
        await prefs.setString('admin_refresh_token', refreshToken);
      }

      final user = await getCurrentAdmin();
      if (user == null || !user.isAdmin) {
        await logout();
        throw StateError('This account does not have admin access');
      }
      return user;
    } catch (error) {
      if (error is StateError) rethrow;
      throw StateError(ApiService.errorMessage(error));
    }
  }

  Future<bool> isLoggedIn() async {
    final prefs = await SharedPreferences.getInstance();
    return (prefs.getString('admin_token') ?? '').isNotEmpty;
  }

  Future<UserModel?> getCurrentAdmin() async {
    if (!await isLoggedIn()) return null;
    try {
      final response = await _apiService.client.get<Map<String, dynamic>>(
        '/auth/me',
      );
      if (response.data == null) return null;
      final user = UserModel.fromJson(response.data!);
      return user.isAdmin ? user : null;
    } catch (error) {
      await logout();
      throw StateError(ApiService.errorMessage(error));
    }
  }

  Future<void> logout() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('admin_token');
    await prefs.remove('admin_refresh_token');
  }
}


===== admin_app/lib/data/services/api_service.dart =====
import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:smart_parking_admin/core/constants/app_constants.dart';

class ApiService {
  ApiService()
      : _dio = Dio(
          BaseOptions(
            baseUrl: AppConstants.apiBaseUrl,
            connectTimeout: const Duration(seconds: 15),
            receiveTimeout: const Duration(seconds: 15),
            headers: const {'Accept': 'application/json'},
          ),
        ) {
    _dio.interceptors.add(
      InterceptorsWrapper(
        onRequest: (options, handler) async {
          final prefs = await SharedPreferences.getInstance();
          final token = prefs.getString('admin_token');
          if (token != null && token.isNotEmpty) {
            options.headers['Authorization'] = 'Bearer $token';
          }
          handler.next(options);
        },
        onError: (error, handler) async {
          if (error.response?.statusCode == 401 &&
              !error.requestOptions.path.endsWith('/auth/login')) {
            final prefs = await SharedPreferences.getInstance();
            await prefs.remove('admin_token');
            await prefs.remove('admin_refresh_token');
          }
          handler.next(error);
        },
      ),
    );
  }

  final Dio _dio;

  Dio get client => _dio;

  static String errorMessage(Object error) {
    if (error is DioException) {
      final data = error.response?.data;
      if (data is Map && data['detail'] != null) {
        return data['detail'].toString();
      }
      if (error.type == DioExceptionType.connectionTimeout ||
          error.type == DioExceptionType.connectionError) {
        return 'Cannot reach the backend at ${AppConstants.apiBaseUrl}. '
            'Start FastAPI on port 8000 and try again.';
      }
      return error.message ?? 'Backend request failed';
    }
    return error.toString();
  }
}

sealed class AuthState {
  const AuthState();
}

class AuthInitial extends AuthState {
  const AuthInitial();
}

class AuthLoading extends AuthState {
  const AuthLoading();
}

class AuthAuthenticated extends AuthState {
  const AuthAuthenticated(this.user);

  final UserModel user;
}

class AuthUnauthenticated extends AuthState {
  const AuthUnauthenticated();
}

class AuthError extends AuthState {
  const AuthError(this.message);

  final String message;
}

class AuthNotifier extends StateNotifier<AuthState> {
  AuthNotifier(this._repository) : super(const AuthInitial()) {
    checkAuthStatus();
  }

  final AuthRepository _repository;

  Future<void> checkAuthStatus() async {
    try {
      final user = await _repository.getCurrentAdmin();
      state = user == null
          ? const AuthUnauthenticated()
          : AuthAuthenticated(user);
    } catch (_) {
      await _repository.logout();
      state = const AuthUnauthenticated();
    }
  }

  Future<bool> login(String email, String password) async {
    state = const AuthLoading();
    try {
      final user = await _repository.login(email, password);
      state = AuthAuthenticated(user);
      return true;
    } catch (error) {
      final message = error
          .toString()
          .replaceFirst('Bad state: ', '')
          .replaceFirst('Exception: ', '');
      state = AuthError(message);
      return false;
    }
  }

  Future<void> logout() async {
    await _repository.logout();
    state = const AuthUnauthenticated();
  }
}

  return AuthNotifier(ref.watch(authRepositoryProvider));

  final state = ref.watch(authStateProvider);
  return state is AuthAuthenticated ? state.user : null;
});

  return ref.watch(authStateProvider) is AuthAuthenticated;

BadgeVariant _variantForStatus(String status) {
  switch (status.toUpperCase()) {
    case 'ACTIVE':
    case 'COMPLETED':
      return BadgeVariant.success;
    case 'PENDING':
    case 'RESERVED':
      return BadgeVariant.warning;
    case 'CANCELLED':
    case 'EXPIRED':
      return BadgeVariant.error;
    default:
      return BadgeVariant.info;
  }
}

import 'dart:math' as math;

import 'package:fl_chart/fl_chart.dart';
import 'package:flutter/material.dart';
import 'package:intl/intl.dart';
import 'package:smart_parking_admin/core/constants/app_colors.dart';
import 'package:smart_parking_admin/core/widgets/admin_card.dart';
import 'package:smart_parking_admin/data/models/stats_model.dart';

class RevenueChart extends StatelessWidget {
  const RevenueChart({super.key, required this.points});

  final List<DailyRevenuePoint> points;

  @override
  Widget build(BuildContext context) {
    final chartPoints = points.isEmpty
        ? List.generate(
            7,
            (index) => DailyRevenuePoint(
              date: DateTime.now().subtract(Duration(days: 6 - index)),
              amount: 0,
            ),
          )
        : points;
    final highest = chartPoints.fold<double>(
      0,
      (value, point) => math.max(value, point.amount),
    );
    final maxY = math.max(1.0, highest * 1.2);

    return AdminCard(
      title: 'Revenue (Last 7 Days)',
      child: SizedBox(
        height: 300,
        child: LineChart(
          LineChartData(
            minX: 0,
            maxX: (chartPoints.length - 1).toDouble(),
            minY: 0,
            maxY: maxY,
            gridData: FlGridData(
              show: true,
              drawVerticalLine: false,
              horizontalInterval: maxY / 4,
            ),
            titlesData: FlTitlesData(
              rightTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              topTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              leftTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  reservedSize: 48,
                  getTitlesWidget: (value, meta) => Text(
                    NumberFormat.compactCurrency(
                      symbol: '₹',
                      decimalDigits: 0,
                    ).format(value),
                    style: Theme.of(context).textTheme.labelSmall,
                  ),
                ),
              ),
              bottomTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  interval: 1,
                  getTitlesWidget: (value, meta) {
                    final index = value.toInt();
                    if (index < 0 || index >= chartPoints.length) {
                      return const SizedBox.shrink();
                    }
                    return Padding(
                      padding: const EdgeInsets.only(top: 8),
                      child: Text(DateFormat.E().format(chartPoints[index].date)),
                    );
                  },
                ),
              ),
            ),
            borderData: FlBorderData(show: false),
            lineBarsData: [
              LineChartBarData(
                spots: [
                  for (var index = 0; index < chartPoints.length; index++)
                    FlSpot(index.toDouble(), chartPoints[index].amount),
                ],
                isCurved: true,
                color: AppColors.primary,
                barWidth: 3,
                isStrokeCapRound: true,
                dotData: FlDotData(show: true),
                belowBarData: BarAreaData(
                  show: true,
                  color: AppColors.primary.withOpacity(0.16),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class _ReservationTab extends ConsumerWidget {
  const _ReservationTab({required this.status});

  final String status;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final reservations = ref.watch(reservationsProvider(status));
    return reservations.when(
      loading: () => const Center(child: CircularProgressIndicator()),
      error: (error, stackTrace) => Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text(
              error.toString().replaceFirst('Bad state: ', ''),
              textAlign: TextAlign.center,
            ),
            TextButton(
              onPressed: () => ref.invalidate(reservationsProvider(status)),
              child: const Text('Retry'),
            ),
          ],
        ),
      ),
      data: (items) => AdminCard(
        title: '${items.length} ${status.toLowerCase()}',
        padding: EdgeInsets.zero,
        child: items.isEmpty
            ? Padding(
                padding: const EdgeInsets.all(32),
                child: Center(
                  child: Text('No ${status.toLowerCase()} reservations'),
                ),
              )
            : SingleChildScrollView(
                scrollDirection: Axis.horizontal,
                child: _ReservationsTable(
                  reservations: items,
                  onCancel: (reservation) =>
                      _cancel(context, ref, reservation),
                ),
              ),
      ),
    );
  }

  Future<void> _cancel(
    BuildContext context,
    WidgetRef ref,
    ReservationModel reservation,
  ) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Cancel reservation?'),
        content: Text(
          'Cancel ${reservation.userName}\'s reservation for '
          '${reservation.slotId}?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Keep'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Cancel reservation'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;

    try {
      await ref
          .read(adminRepositoryProvider)
          .forceCancelReservation(reservation.id);
      ref.invalidate(reservationsProvider(status));
      ref.invalidate(reservationsProvider(''));
      ref.invalidate(dashboardStatsProvider);
    } catch (error) {
      if (context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(error.toString().replaceFirst('Bad state: ', '')),
          ),
        );
      }
    }
  }
}

class _ReservationsTable extends StatelessWidget {
  const _ReservationsTable({
    required this.reservations,
    required this.onCancel,
  });

  final List<ReservationModel> reservations;
  final ValueChanged<ReservationModel> onCancel;

  @override
  Widget build(BuildContext context) {
    final currency = NumberFormat.currency(
      locale: 'en_IN',
      symbol: '₹',
      decimalDigits: 0,
    );
    return DataTable(
      columns: const [
        DataColumn(label: Text('User')),
        DataColumn(label: Text('Slot')),
        DataColumn(label: Text('Area')),
        DataColumn(label: Text('Start time')),
        DataColumn(label: Text('Status')),
        DataColumn(label: Text('Amount')),
        DataColumn(label: Text('Action')),
      ],
      rows: [
        for (final reservation in reservations)
          DataRow(
            cells: [
              DataCell(Text(reservation.userName)),
              DataCell(Text(reservation.slotId)),
              DataCell(Text(reservation.areaName)),
              DataCell(
                Text(
                  DateFormat('dd MMM yyyy, hh:mm a')
                      .format(reservation.startTime.toLocal()),
                ),
              ),
              DataCell(
                StatusBadge(
                  text: reservation.status,
                  variant: _statusVariant(reservation.status),
                ),
              ),
              DataCell(Text(currency.format(reservation.amount))),
              DataCell(
                TextButton(
                  onPressed: reservation.canCancel
                      ? () => onCancel(reservation)
                      : null,
                  child: const Text('Cancel'),
                ),
              ),
            ],
          ),
      ],
    );
  }
}

BadgeVariant _statusVariant(String status) {
  switch (status) {
    case 'ACTIVE':
    case 'COMPLETED':
      return BadgeVariant.success;
    case 'PENDING':
      return BadgeVariant.warning;
    case 'CANCELLED':
    case 'EXPIRED':
      return BadgeVariant.error;
    default:
      return BadgeVariant.info;
  }
}

                ),
              ],
            ),

                            ],

                                _buildRow(
                                  context,
                                  ref,

                                ),
                            ],
                          ),
                        ),
                ),
              ),
            ),
          ],
        ),
      ),

  DataRow _buildRow(
    BuildContext context,
    WidgetRef ref,
    UserModel user, {
    required bool isCurrentUser,
  }) {
    return DataRow(
      cells: [
        DataCell(Text(user.name)),
        DataCell(Text(user.email)),
        DataCell(StatusBadge(text: user.role.name, variant: BadgeVariant.info)),
        DataCell(
          StatusBadge(
            text: user.isActive ? 'Active' : 'Suspended',
            variant: user.isActive
                ? BadgeVariant.success
                : BadgeVariant.error,
